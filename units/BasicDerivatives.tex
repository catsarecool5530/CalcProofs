\subsection{Addition Rule}

Should be trivial, is trivial, most people don't even think this needs a proof....

\begin{proof}
\begin{align}
\frac{d}{dx} f(x) + g(x) 
&= \lim_{h \to 0} \frac{f(x + h) + g(x + h) - f(x) - g(x)}{h}
&= \lim_{h \to 0} \frac{f(x + h) - f(x)}{h} + \lim_{h \to 0} \frac{g(x + h) - g(x)}{h}
&= f'(x) + g'(x)
\end{align}
\end{proof}
\subsection{Power Rule for $n \in \mathbb{N}$}
Easy proof with binomial theorem.
\begin{proof}
consider $f(x) = x^n$, $n \in \mathbb{N}$.

\begin{align}
\frac{d}{dx} x^n &= \lim_{h \to 0} \frac{(x + h)^n - x^n}{h} \\
&= \lim_{h \to 0} \frac{-x^n + \binom{0}{n}x^n + \binom{1}{n}hx^{n-1} + \binom{2}{n}h^2n^{n-2} + \ldots + \binom{n-2}h^{n-2}n^2 + \binom{n-1}{n}h^{n-1}n + \binom{n}{n} h^n}{h} \\
&= \lim_{h \to 0} \frac{-x^n + x^n + nhx^{n-1} + \frac{n(n-1)}{2}h^2n^{n-2} + \ldots + \frac{n(n-1)}{2}h^{n-2}n^2 + nh^{n-1}n + h^n}{h} \\
&= \lim_{h \to 0} \frac{nhx^{n-1} + \frac{n(n-1)}{2}h^2n^{n-2} + \ldots + \frac{n(n-1)}{2}h^{n-2}n^2 + nh^{n-1}n + h^n}{h} \\ 
&= \lim_{h \to 0} \frac{h(nx^{n-1} + \frac{n(n-1)}{2}hn^{n-2} + \ldots + \frac{n(n-1)}{2}h^{n-3}n^2 + nh^{n-2}n + h^{n-1}}{h} \\ 
&= \lim_{h \to 0} nx^{n-1} + \frac{n(n-1)}{2}hn^{n-2} + \ldots + \frac{n(n-1)}{2}h^{n-3}n^2 + nh^{n-2}n + h^{n-1} \\
&= nx^{n-1}
\end{align}
\end{proof}

\subsection{Basic Trig}

Here is $\cos x$:
\begin{proof}
Remember that:

\[\cos(\alpha + \beta) = \cos\alpha\cos\beta - \sin\alpha\sin\beta\]

Now consider our def of limit:

\begin{align}
&\lim_{h \to 0} \frac{\cos(x + h) - \cos(x)}{h} \\
&= \lim_{h \to 0} \frac{\cos x \cos h - \sin x \sin h - \cos x}{h} \\
&= \lim_{h \to 0} \frac{\cos x(\cos h - 1) - \sin x \sin h}{h} \\ 
&= \lim_{h \to 0} \frac{\cos x(\cos h - 1)}{h} - \lim_{h \to 0} \frac{\sin x \sin h}{h}
\end{align}

Now, 
\[\lim_{h \to 0} \frac{\cos h - 1}{h} = 0\]
and
\[\lim_{h \to 0} \frac{\sin h}{h} = 1\]

so it simplifies down to

$\cos x \cdot 0 - \sin x \cdot 1 =$ \fbox{-sin x}
\end{proof}

\subsection{Quotient Rule}


Quotient rule is pretty easy too. No help needed.
\begin{proof}
\begin{align}
\frac{d}{dx} \frac{f(x)}{g(x)}
&= \lim_{h \to 0} \frac{\frac{f(x + h)}{g(x + h} - \frac{f(x)}{g(x)}}{h} \\
&= \lim_{h \to 0} \frac{f(x + h)g(x) - f(x)g(x + h)}{h \cdot g(x)g(x + h)} \\
&= \lim_{h \to 0} \frac{1}{g(x)g(x + h)} \cdot \frac{f(x + h)g(x) - f(x)g(x) + f(x)g(x) - f(x)g(x + h)}{h} \\ 
&= \lim_{h \to 0} \frac{1}{g(x)g(x + h)} \cdot \frac{g(x)(f(x + h) - f(x)) + f(x)(g(x) - g(x + h))}{h} \\
&= \lim_{h \to 0} \frac{1}{g(x)g(x + h)} \cdot (g(x) \frac{f(x + h) - f(x)}{h} - f(x) \frac{g(x + h) - g(x)}{h})\\
&= \lim_{h \to 0} \frac{1}{g(x)g(x + h)} \cdot (g(x) f'(x) - f(x) g'(x))\\
&= \frac{g(x) f'(x) - f(x) g'(x)}{g(x)^2}
\end{align}
\end{proof}


\subsection{Power Rule for $n \in \mathbb{Q}$}

This one is solved in the textbook with implicit differentiation, but I provide my own proof here.

\begin{proof}
Recall that 
\[a^n - b^n = (a - b)(a^{n-1} + ba^{n-2} + b^2a^{n-3} + \ldots +  b^{n-2}a + b^{n-1})\]
The proof is trivial by "just distribute it"

Consider 
\[x^{\frac{1}{n}}, n \in \mathbb{N}\]

\begin{align}
\frac{d}{dx} x ^{\frac{1}{n}} &= \lim_{h \to 0} \frac{(x + h)^{\frac{1}{n}} - x^{\frac{1}{n}}}{h} \\
&= \lim_{h \to 0} \frac{(x + h)^{\frac{1}{n}} - x^{\frac{1}{n}}}{h} \cdot \frac{(x+h)^{\frac{n-1}{n}} + (x+h)^{\frac{n-2}{n}} x^{\frac{1}{n}} + \ldots + (x+h)^{\frac{1}{n}} x ^ {\frac{n - 2}{n}} + x^{\frac{n-1}{n}}}{(x+h)^{\frac{n-1}{n}} + (x+h)^{\frac{n-2}{n}} x^{\frac{1}{n}}  + \ldots + (x+h)^{\frac{1}{n}} x ^ {\frac{n - 2}{n}} + x^{\frac{n-1}{n}}} \\
&=\lim_{h \to 0} \frac{x + h - x}{h((x+h)^{\frac{n-1}{n}} + (x+h)^{\frac{n-2}{n}} x^{\frac{1}{n}}  + \ldots + (x+h)^{\frac{1}{n}} x ^ {\frac{n - 2}{n}} + x^{\frac{n-1}{n}})}\\
&= \lim_{h \to 0}\frac{h}{h((x+h)^{\frac{n-1}{n}} + (x+h)^{\frac{n-2}{n}} x^{\frac{1}{n}}  + \ldots + (x+h)^{\frac{1}{n}} x ^ {\frac{n - 2}{n}} + x^{\frac{n-1}{n}})}\\
&= \lim_{h \to 0}\frac{1}{((x+h)^{\frac{n-1}{n}} + (x+h)^{\frac{n-2}{n}} x^{\frac{1}{n}}  + \ldots + (x+h)^{\frac{1}{n}} x ^ {\frac{n - 2}{n}} + x^{\frac{n-1}{n}})}
\end{align}

Now we can plug in 0 to get 
\begin{align}
&\frac{1}{nx^{\frac{n-1}{n}}}\\
&= \frac{1}{n} x^{\frac{1}{n} - 1}
\end{align}

Thus, the power rule holds for all nth roots.

Next, consider a function of the form 
\[x^{\frac{a}{b}} (a, b \in \mathbb{N})\]

\begin{align}
x^{\frac{a}{b}} \\ 
&= (x^a)^{\frac{1}{b}}
\end{align}
Applying chain rule
\begin{align}
\frac{d}{dx}((x^a)^{\frac{1}{b}}) &= \frac{1}{b}(x^a)^{\frac{1}{b} - 1} a x^{a-1} \\
&= \frac{a}{b} x^{\frac{a}{b} - a} x^{a - 1} \\
&= \frac{a}{b} x^{\frac{a}{b} -a  + a - 1}\\
\frac{d}{dx} x^{\frac{a}{b}}&= \frac{a}{b} x^{\frac{a}{b} - 1}
\end{align}

Thus, the power rule holds for all rational powers.
\end{proof}

Regardless, I am going to provide the implicit differentiation proof here for reference (didnt need help from it though):

\begin{proof}
Consider $y = x^{\frac{a}{b}}$, where $a, b \in \mathbb{N}$
\begin{align}
y^b &= (x^{\frac{a}{b}})^b \\
y^b &= x^a \\
b y^{b-1} \frac{dy}{dx} &= a x^{a-1} \\
\frac{dy}{dx} &= \frac{a x^{a-1}}{b y^{b-1}} \\
&= \frac{a x^{a-1}}{b (x^{\frac{a}{b}})^{b-1}} \\
&= \frac{a x^{a-1}}{b x^{a - \frac{a}{b}}} \\
&= \frac{a}{b} x^{a - 1 - a + \frac{a}{b}} \\
&= \frac{a}{b} x^{\frac{a}{b} - 1}
\end{align}
\end{proof}

Pretty simple, which is why the textbook used it.
